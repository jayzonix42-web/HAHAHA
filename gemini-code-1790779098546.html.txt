<script>
  let usdBalance = 856384831.36;
  const coinRate = 0.012137;

  let selectedCoins = 250;
  let selectedCost = 3.03;

  // DOM Elements
  const balanceUsdEl = document.getElementById('balance-usd');
  const balanceCoinsEl = document.getElementById('balance-coins');
  const balanceSubUsdEl = document.getElementById('balance-sub-usd');
  
  const userInput = document.getElementById('user-input');
  const profileImg = document.getElementById('profile-img');
  const profileHandle = document.getElementById('profile-handle');

  const coinCards = document.querySelectorAll('.coin-card');
  const customContainer = document.getElementById('custom-container');
  const customCoinsInput = document.getElementById('custom-coins-input');
  const exchangeBtn = document.getElementById('exchange-btn');

  const processingScreen = document.getElementById('processing-screen');
  const receiptScreen = document.getElementById('receipt-screen');
  const receiptBackBtn = document.getElementById('receipt-back-btn');

  function updateBalanceDisplay() {
    const formattedUsd = '$' + usdBalance.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    balanceUsdEl.textContent = formattedUsd;
    balanceSubUsdEl.textContent = formattedUsd;
    
    const totalCoins = Math.floor(usdBalance / coinRate);
    balanceCoinsEl.textContent = totalCoins.toLocaleString('en-US');
  }

  // Parse custom coin inputs like "10k", "1.5k", "1m", or "10,000"
  function parseCoinInput(value) {
    if (!value) return 0;
    let clean = value.toString().trim().toLowerCase().replace(/,/g, '');
    
    let multiplier = 1;
    if (clean.endsWith('k')) {
      multiplier = 1000;
      clean = clean.slice(0, -1);
    } else if (clean.endsWith('m')) {
      multiplier = 1000000;
      clean = clean.slice(0, -1);
    } else if (clean.endsWith('b')) {
      multiplier = 1000000000;
      clean = clean.slice(0, -1);
    }
    
    const parsed = parseFloat(clean);
    return isNaN(parsed) ? 0 : Math.floor(parsed * multiplier);
  }

  // Username Input Handler
  userInput.addEventListener('input', (e) => {
    const username = e.target.value.trim() || 'user';
    profileHandle.textContent = '@' + username;
    profileImg.src = `https://unavatar.io/tiktok/${username}`;
  });

  // Coin Grid Card Handler
  coinCards.forEach(card => {
    card.addEventListener('click', () => {
      coinCards.forEach(c => c.classList.remove('selected'));
      customContainer.classList.remove('selected');
      customCoinsInput.value = '';

      card.classList.add('selected');
      selectedCoins = parseInt(card.dataset.coins);
      selectedCost = parseFloat(card.dataset.cost);
    });
  });

  // Custom Input Handlers
  customCoinsInput.addEventListener('input', (e) => {
    coinCards.forEach(c => c.classList.remove('selected'));
    customContainer.classList.add('selected');

    const rawValue = e.target.value;
    const coins = parseCoinInput(rawValue);
    selectedCoins = coins;
    selectedCost = parseFloat((coins * coinRate).toFixed(2));
  });

  customCoinsInput.addEventListener('blur', (e) => {
    const coins = parseCoinInput(e.target.value);
    if (coins > 0) {
      e.target.value = coins.toLocaleString('en-US');
    }
  });

  // Exchange Action with Processing Step
  exchangeBtn.addEventListener('click', () => {
    if (selectedCoins <= 0) {
      alert("Please enter or select a valid coin amount.");
      return;
    }

    if (usdBalance < selectedCost) {
      alert("Insufficient Balance");
      return;
    }

    // 1. Show Processing Recharge Screen
    processingScreen.style.display = 'flex';

    // 2. Wait 2 seconds before completing exchange
    setTimeout(() => {
      // Hide Processing Screen
      processingScreen.style.display = 'none';

      // Deduct balance
      usdBalance -= selectedCost;
      updateBalanceDisplay();

      // Populate Receipt Details
      document.getElementById('rec-coins').textContent = selectedCoins.toLocaleString('en-US');
      document.getElementById('rec-coins-detail').textContent = selectedCoins.toLocaleString('en-US');
      document.getElementById('rec-user').textContent = profileHandle.textContent;
      document.getElementById('rec-cost').textContent = selectedCost.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
      
      const now = new Date();
      document.getElementById('rec-time').textContent = now.toLocaleString();

      // 3. Show Completed Receipt Screen
      receiptScreen.style.display = 'flex';
    }, 2000);
  });

  // Back Button Handler
  receiptBackBtn.addEventListener('click', () => {
    receiptScreen.style.display = 'none';
  });

  // Initial Sync
  updateBalanceDisplay();
</script>
